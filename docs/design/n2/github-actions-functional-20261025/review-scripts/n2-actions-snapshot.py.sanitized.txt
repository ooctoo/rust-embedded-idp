# Redacted audit source transcript; private machine paths are placeholders.
import concurrent.futures,datetime,json,runpy,sys
c=runpy.run_path('/tmp/n2-first-actions-control.py'); api=c['api']; out=c['PUBLIC']
run_id=int(sys.argv[1]) if len(sys.argv)>1 else json.loads((out/'first-run-initial.json').read_text())['id']
root=out/'runs'/str(run_id); root.mkdir(parents=True,exist_ok=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
    a=pool.submit(api,'actions/runs/'+str(run_id)); b=pool.submit(api,'actions/runs/'+str(run_id)+'/jobs?filter=all&per_page=100')
    run=a.result(); jobs=b.result()
(root/'run-latest.json').write_text(json.dumps(run,indent=2)+'\n'); (root/'jobs-latest.json').write_text(json.dumps(jobs,indent=2)+'\n')
state={'runId':run_id,'headSha':run['head_sha'],'event':run['event'],'status':run['status'],'conclusion':run['conclusion'],'attempt':run['run_attempt'],'jobs':[{'id':j['id'],'status':j['status'],'conclusion':j['conclusion'],'labels':j['labels'],'runner':j['runner_name'],'steps':[{'name':s['name'],'status':s['status'],'conclusion':s['conclusion']} for s in j['steps'] if s['status']!='pending']} for j in jobs['jobs']]}
p=root/'timeline.json'; history=json.loads(p.read_text()) if p.exists() else []
if not history or history[-1]['state']!=state:
    history.append({'observedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'state':state}); p.write_text(json.dumps(history,indent=2)+'\n')
print(json.dumps(state,ensure_ascii=False))
